/**
 * Nimbo Data Ingester Service

 * This is the main entry point for the Nimbo data processing pipeline.

 * and writes them as Parquet files to HDFS.
 *

package com.nimbo.ingester.config;

import com.nimbo.ingester.deserializer.AvroDeserializer;
import org.apache.avro.Schema;
import org.springframework.boot.context.properties.EnableConfigurationProperties;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.core.io.ClassPathResource;

import java.io.IOException;
import java.io.InputStream;
import java.util.HashMap;
import java.util.Map;
import org.apache.kafka.clients.producer.KafkaProducer;
import org.apache.kafka.clients.producer.ProducerConfig;
import org.apache.kafka.common.serialization.ByteArraySerializer;

@Configuration
@EnableConfigurationProperties(KafkaProperties.class)
public class IngesterConfiguration {

    private static final String AVRO_SCHEMA_PATH = "avro/NetFlowV2Record.avsc";

    @Bean
    public Schema avroSchema() {
        try (InputStream schemaStream =
                     new ClassPathResource(AVRO_SCHEMA_PATH).getInputStream()) {

            return new Schema.Parser().parse(schemaStream);

        } catch (IOException e) {
            throw new IllegalStateException(
                    "Failed to load Avro schema from: " + AVRO_SCHEMA_PATH, e
            );
        }
    }

    @Bean
    public Schema storageSchema(Schema avroSchema) {
        Schema.Field processingTimeField = new Schema.Field(
                "PROCESSING_TIME",
                Schema.createUnion(
                        Schema.create(Schema.Type.NULL),
                        Schema.create(Schema.Type.LONG)
                ),
                "Time when the record was ingested into HDFS (UTC epoch microseconds).",
                (Object) null
        );

        java.util.List<Schema.Field> fields = new java.util.ArrayList<>();
        for (Schema.Field field : avroSchema.getFields()) {
            Schema.Field copy = new Schema.Field(
                    field.name(),
                    field.schema(),
                    field.doc(),
                    field.defaultVal()
            );
            fields.add(copy);
        }

        fields.add(processingTimeField);

        return Schema.createRecord(
                avroSchema.getName(),
                avroSchema.getDoc(),
                avroSchema.getNamespace(),
                avroSchema.isError(),
                fields
        );
    }

    @Bean
    public AvroDeserializer avroDeserializer(Schema avroSchema) {
        return new AvroDeserializer(avroSchema);
    }

    @Bean(destroyMethod = "close")
    public KafkaProducer<byte[], byte[]> kafkaProducer(
            KafkaProperties properties
    ) {

        Map<String, Object> configs =
                new HashMap<>();

        configs.put(
                ProducerConfig.BOOTSTRAP_SERVERS_CONFIG,
                properties.getBootstrapServers()
        );

        configs.put(
                ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG,
                ByteArraySerializer.class.getName()
        );
        configs.put(
                ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG,
                ByteArraySerializer.class.getName()
        );

        configs.put(
                ProducerConfig.ACKS_CONFIG,
                "all"
        );

        configs.put(
                ProducerConfig.ENABLE_IDEMPOTENCE_CONFIG,
                true
        );

        return new KafkaProducer<>(configs);
    }
}

package com.nimbo.ingester.consumer;

import com.nimbo.ingester.config.KafkaProperties;
import com.nimbo.ingester.deserializer.AvroDeserializer;
import com.nimbo.ingester.service.DataProcessingService;
import com.nimbo.ingester.service.DlqService;
import com.nimbo.ingester.service.NetflowValidationService;
import com.nimbo.metrics.BatchIngesterMetrics;
import com.nimbo.storage.config.HdfsProperties;
import com.nimbo.storage.config.ParquetProperties;
import com.nimbo.storage.service.NetFlowStorage;
import com.nimbo.storage.service.batching.BatchPolicy;
import com.nimbo.storage.service.hdfs.HdfsFileStorage;
import com.nimbo.storage.service.naming.FileNameGenerator;
import com.nimbo.storage.service.parquet.ParquetFileWriter;
import jakarta.annotation.PostConstruct;
import jakarta.annotation.PreDestroy;
import lombok.extern.slf4j.Slf4j;
import org.apache.hadoop.fs.FileSystem;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import com.nimbo.storage.service.hdfs.HdfsFileSystemFactory;
import org.springframework.stereotype.Component;
import org.apache.avro.Schema;

import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

@Slf4j
@Component
public class ConsumerManager {

    private final KafkaProperties kafkaProperties;
    private final AvroDeserializer avroDeserializer;
    private final NetflowValidationService validationService;
    private final DlqService dlqService;
    private final BatchPolicy batchPolicy;
    private final ParquetProperties parquetProperties;
    private final FileNameGenerator fileNameGenerator;
    private final HdfsProperties hdfsProperties;
    private final KafkaConsumerFactory kafkaConsumerFactory;
    private final HdfsFileSystemFactory hdfsFileSystemFactory;
    private final KafkaOffsetCommitter kafkaOffsetCommitter;
    private final Schema storageSchema;
    private final BatchIngesterMetrics metrics;

    private final List<NetflowConsumer> consumers = new ArrayList<>();
    private ExecutorService consumerExecutor;

    public ConsumerManager(
            KafkaProperties kafkaProperties,
            AvroDeserializer avroDeserializer,
            NetflowValidationService validationService,
            DlqService dlqService,
            BatchPolicy batchPolicy,
            ParquetProperties parquetProperties,
            FileNameGenerator fileNameGenerator,
            HdfsProperties hdfsProperties,
            KafkaConsumerFactory kafkaConsumerFactory,
            HdfsFileSystemFactory hdfsFileSystemFactory,
            KafkaOffsetCommitter kafkaOffsetCommitter,
            Schema storageSchema,
            BatchIngesterMetrics metrics) {
        this.kafkaProperties = kafkaProperties;
        this.avroDeserializer = avroDeserializer;
        this.validationService = validationService;
        this.dlqService = dlqService;
        this.batchPolicy = batchPolicy;
        this.parquetProperties = parquetProperties;
        this.fileNameGenerator = fileNameGenerator;
        this.hdfsProperties = hdfsProperties;
        this.kafkaConsumerFactory = kafkaConsumerFactory;
        this.hdfsFileSystemFactory = hdfsFileSystemFactory;
        this.kafkaOffsetCommitter = kafkaOffsetCommitter;
        this.storageSchema = storageSchema;
        this.metrics = metrics;
    }

    @PostConstruct
    public void start() {
        int instanceCount = kafkaProperties.getConsumerInstances();
        log.info("Starting {} consumer instances for topic: {}", instanceCount, kafkaProperties.getTopic());

        consumerExecutor = Executors.newFixedThreadPool(
                instanceCount,
                r -> {
                    Thread t = new Thread(r);
                    t.setName("consumer-instance-" + t.getId());
                    t.setDaemon(true);
                    return t;
                }
        );

        for (int i = 0; i < instanceCount; i++) {

            int instanceId = i + 1;

            try {

                metrics.registerInstance(instanceId);
                FileSystem fs = hdfsFileSystemFactory.create(hdfsProperties);
                HdfsFileStorage fileStorage = new HdfsFileStorage(fs);
                ParquetFileWriter parquetWriter = new ParquetFileWriter(fs, parquetProperties);

                NetFlowStorage storage = new NetFlowStorage(
                        batchPolicy,
                        parquetWriter,
                        fileStorage,
                        fileNameGenerator,
                        hdfsProperties,
                        storageSchema,
                        metrics,
                        instanceId
                );

                DataProcessingService processingService = new DataProcessingService(storage);

                KafkaConsumer<byte[], byte[]> kafkaConsumer =
                        kafkaConsumerFactory.createConsumer(i + 1);
                metrics.bindKafkaConsumer(kafkaConsumer);

                NetflowConsumer consumer = new NetflowConsumer(
                        kafkaConsumer,
                        kafkaProperties,
                        avroDeserializer,
                        processingService,
                        validationService,
                        dlqService,
                        kafkaOffsetCommitter,
                        instanceId,
                        metrics
                );

                consumers.add(consumer);
                consumerExecutor.submit(consumer::start);
                log.info("Started consumer instance #{}", i + 1);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                log.error("Thread interrupted while creating consumer instance #{}", i + 1, e);
                break;

            } catch (IOException | RuntimeException e) {
                log.error("Failed to create consumer instance #{}", i + 1, e);
            }
        }
    }

    @PreDestroy
    public void shutdown() {
        log.info("Shutting down ConsumerManager...");

        for (NetflowConsumer consumer : consumers) {
            consumer.shutdown();
        }

        if (consumerExecutor != null) {
            consumerExecutor.shutdown();
            try {
                if (!consumerExecutor.awaitTermination(30, TimeUnit.SECONDS)) {
                    consumerExecutor.shutdownNow();
                }
            } catch (InterruptedException e) {
                consumerExecutor.shutdownNow();
                Thread.currentThread().interrupt();
            }
        }

        log.info("ConsumerManager shutdown complete");
    }
}

package com.nimbo.ingester.consumer;

import com.nimbo.storage.model.TopicPartitionKey;
import lombok.extern.slf4j.Slf4j;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import org.apache.kafka.clients.consumer.OffsetAndMetadata;
import org.apache.kafka.common.TopicPartition;
import org.springframework.stereotype.Component;

import java.util.HashMap;
import java.util.Map;

@Component
@Slf4j
public class KafkaOffsetCommitter {

    public void commit(KafkaConsumer<byte[], byte[]> kafkaConsumer,
                       Map<TopicPartitionKey, Long> offsets) {
        if (offsets == null || offsets.isEmpty()) {
            return;
        }

        Map<TopicPartition, OffsetAndMetadata> commitMap = new HashMap<>();
        offsets.forEach((key, offset) -> {
            TopicPartition partition = new TopicPartition(key.topic(), key.partition());
            commitMap.put(partition, new OffsetAndMetadata(offset));
        });

         kafkaConsumer.commitSync(commitMap);
    }
}

package com.nimbo.ingester.consumer;

import com.nimbo.ingester.config.KafkaProperties;
import com.nimbo.ingester.deserializer.AvroDeserializer;
import com.nimbo.ingester.exception.DeserializationException;
import com.nimbo.ingester.exception.KafkaConsumerException;
import com.nimbo.ingester.model.IngestedRecord;
import com.nimbo.ingester.model.ValidationResult;
import com.nimbo.ingester.service.DataProcessingService;
import com.nimbo.ingester.service.DlqService;
import com.nimbo.ingester.service.NetflowValidationService;
import com.nimbo.metrics.BatchIngesterMetrics;
import com.nimbo.storage.exception.StorageException;
import com.nimbo.storage.model.TopicPartitionKey;
import lombok.extern.slf4j.Slf4j;
import org.apache.kafka.clients.consumer.ConsumerRebalanceListener;
import org.apache.kafka.clients.consumer.ConsumerRecord;
import org.apache.kafka.clients.consumer.ConsumerRecords;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import org.apache.kafka.common.KafkaException;
import org.apache.kafka.common.TopicPartition;

import java.time.Duration;
import java.util.*;
import java.util.concurrent.atomic.AtomicBoolean;

@Slf4j
public class NetflowConsumer {

    private final KafkaConsumer<byte[], byte[]> kafkaConsumer;
    private final KafkaProperties kafkaProperties;
    private final AvroDeserializer avroDeserializer;
    private final DataProcessingService dataProcessingService;
    private final NetflowValidationService validationService;
    private final DlqService dlqService;
    private final int instanceId;
    private final KafkaOffsetCommitter kafkaOffsetCommitter;

    private final AtomicBoolean running = new AtomicBoolean(true);
    private Thread consumerThread;
    private final BatchIngesterMetrics metrics;

    public NetflowConsumer(
            KafkaConsumer<byte[], byte[]> kafkaConsumer,
            KafkaProperties kafkaProperties,
            AvroDeserializer avroDeserializer,
            DataProcessingService dataProcessingService,
            NetflowValidationService validationService,
            DlqService dlqService,
            KafkaOffsetCommitter kafkaOffsetCommitter,
            int instanceId,BatchIngesterMetrics metrics) {
        this.kafkaConsumer = kafkaConsumer;
        this.kafkaProperties = kafkaProperties;
        this.avroDeserializer = avroDeserializer;
        this.dataProcessingService = dataProcessingService;
        this.validationService = validationService;
        this.dlqService = dlqService;
        this.kafkaOffsetCommitter = kafkaOffsetCommitter;
        this.instanceId = instanceId;
        this.metrics = metrics;
    }

    private void registerRebalanceListener() {
        this.kafkaConsumer.subscribe(
                Collections.singletonList(kafkaProperties.getTopic()),
                new ConsumerRebalanceListener() {
                    @Override
                    public void onPartitionsRevoked(Collection<TopicPartition> partitions) {
                        log.info("Instance #{}: Partitions revoked: {}", instanceId, partitions);
                        try {
                            Map<TopicPartitionKey, Long> flushedOffsets = dataProcessingService.flush();
                            kafkaOffsetCommitter.commit(kafkaConsumer, flushedOffsets);
                            log.info("Instance #{}: Flushed {} offsets during rebalance",
                                    instanceId, flushedOffsets.size());
                        } catch (StorageException | KafkaException e) {
                            log.error("Instance #{}: Failed to flush during rebalance", instanceId, e);
                        }
                    }

                    @Override
                    public void onPartitionsAssigned(Collection<TopicPartition> partitions) {
                        log.info("Instance #{}: Partitions assigned: {}", instanceId, partitions);
                    }
                }
        );
    }

    public void start() {
        log.info("Starting consumer instance #{} for topic: {}",
                instanceId, kafkaProperties.getTopic());
        this.consumerThread = new Thread(this::consumeLoop,
                "consumer-instance-" + instanceId);
        this.consumerThread.setDaemon(true);
        this.consumerThread.start();
    }

    private void consumeLoop() {
        try {
            registerRebalanceListener();
            metrics.consumerStarted(instanceId);
            log.info("Instance #{}: Subscribed to topic: {}", instanceId, kafkaProperties.getTopic());

            while (running.get()) {
                consumePoll();
            }

        } catch (KafkaException | IllegalArgumentException | IllegalStateException e) {
            log.error("Instance #{}: Fatal error in consumer loop", instanceId, e);
            throw new KafkaConsumerException("Consumer loop failed", e);
        } finally {
            metrics.consumerStopped(instanceId);
            closeConsumer();
        }
    }

    private void consumePoll() {
        try {
            ConsumerRecords<byte[], byte[]> records = kafkaConsumer.poll(
                    Duration.ofMillis(kafkaProperties.getPollTimeoutMs())
            );

            metrics.recordsPolled(instanceId, records.count());
            if (records.isEmpty()) {
                handleEmptyPoll();
                return;
            }

            Map<TopicPartitionKey, Long> offsetsToCommit = processRecords(records);
            commitOffsets(offsetsToCommit);

        } catch (KafkaException e) {
            log.error("Instance #{}: Kafka error during polling or offset commit", instanceId, e);
            throw new KafkaConsumerException("Kafka polling failed for consumer instance #" + instanceId, e);

        } catch (StorageException | IllegalStateException e) {
            log.error("Instance #{}: Error while processing or flushing consumed records", instanceId, e);
        }
    }

    private void handleEmptyPoll() {
        log.debug("No records received from Kafka");

        Map<TopicPartitionKey, Long> offsetsToCommit =
                dataProcessingService.flushRemaining();

        commitOffsets(offsetsToCommit);
    }

    private Map<TopicPartitionKey, Long> processRecords(
            ConsumerRecords<byte[], byte[]> records) {

        Map<TopicPartitionKey, Long> offsetsToCommit = new HashMap<>();

        for (ConsumerRecord<byte[], byte[]> consumerRecord : records) {
            processRecord(consumerRecord).ifPresent(offsets ->
                    offsets.forEach((key, offset) ->
                            offsetsToCommit.merge(key, offset, Math::max)));
        }

        return offsetsToCommit;
    }

    private Optional<Map<TopicPartitionKey, Long>> processRecord(ConsumerRecord<byte[], byte[]> consumerRecord) {
        long nextOffset = consumerRecord.offset() + 1;
        try {
            var genericRecord = avroDeserializer.deserialize(consumerRecord.value());
            ValidationResult result = validationService.validate(genericRecord);

            if (!result.valid()) {
                metrics.invalidRecord(instanceId, "VALIDATION_ERROR");
                dlqService.publish(consumerRecord, "VALIDATION_ERROR", result.errorMessage());
                return dataProcessingService.recordSkippedOffset(
                        consumerRecord.topic(), consumerRecord.partition(), nextOffset);
            }

            IngestedRecord ingestedRecord = new IngestedRecord(
                    genericRecord,
                    consumerRecord.topic(),
                    consumerRecord.partition(),
                    consumerRecord.offset(),
                    consumerRecord.value().length
            );

            return dataProcessingService.processRecord(ingestedRecord);

        } catch (DeserializationException e) {
            metrics.invalidRecord(instanceId, "DESERIALIZATION_ERROR");
            dlqService.publish(consumerRecord, "DESERIALIZATION_ERROR", e.getMessage());
            return dataProcessingService.recordSkippedOffset(
                    consumerRecord.topic(), consumerRecord.partition(), nextOffset);
        } catch (StorageException e) {
            log.error("Instance #{}: Error processing record: topic={}, partition={}, offset={}",
                    instanceId, consumerRecord.topic(), consumerRecord.partition(), consumerRecord.offset(), e);
            return Optional.empty();
        }
    }

    private void commitOffsets(Map<TopicPartitionKey, Long> offsetsToCommit) {
        if (kafkaProperties.isEnableAutoCommit() || offsetsToCommit.isEmpty()) {
            return;
        }

        kafkaOffsetCommitter.commit(kafkaConsumer, offsetsToCommit);

        log.debug("Instance #{}: Committed offsets: {}",
                instanceId, offsetsToCommit);
    }

    private void closeConsumer() {
        try {
            kafkaConsumer.close();
            log.info("Instance #{}: Kafka consumer closed", instanceId);
        } catch (KafkaException | IllegalStateException e) {
            log.warn("Instance #{}: Error closing Kafka consumer", instanceId, e);
        }
    }

    public void shutdown() {
        log.info("Instance #{}: Shutting down...", instanceId);
        running.set(false);

        try {
            log.info("Flushing remaining buffered records before shutdown...");
            Map<TopicPartitionKey, Long> flushedOffsets = dataProcessingService.flush();

            if (!flushedOffsets.isEmpty()) {
                kafkaOffsetCommitter.commit(kafkaConsumer, flushedOffsets);
                log.info("Instance #{}: Flushed {} offsets during shutdown",
                        instanceId, flushedOffsets.size());
            }

        } catch (StorageException | KafkaException | IllegalStateException e) {
            log.error("Instance #{}: Error flushing during shutdown", instanceId, e);
        }

        if (consumerThread != null) {
            try {
                consumerThread.interrupt();
                consumerThread.join(5000);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }

        log.info("Instance #{}: Shutdown complete", instanceId);
    }
}

package com.nimbo.ingester.deserializer;

import com.nimbo.ingester.exception.DeserializationException;
import lombok.Getter;
import org.apache.avro.Schema;
import org.apache.avro.generic.GenericDatumReader;
import org.apache.avro.generic.GenericRecord;
import org.apache.avro.io.BinaryDecoder;
import org.apache.avro.io.DecoderFactory;

import java.io.IOException;

public class AvroDeserializer {

    @Getter
    private final Schema schema;
    private final GenericDatumReader<GenericRecord> reader;
    private final DecoderFactory decoderFactory = DecoderFactory.get();

    public AvroDeserializer(Schema schema) {
        this.schema = schema;
        this.reader = new GenericDatumReader<>(schema);
    }

    public GenericRecord deserialize(byte[] data) {
        try {
            BinaryDecoder decoder = decoderFactory.binaryDecoder(data, null);
            return reader.read(null, decoder);
        } catch (IOException | IllegalArgumentException e) {
            throw new DeserializationException("Failed to deserialize Avro record", e);
        }
    }

}

import org.apache.avro.generic.GenericRecord;

import java.util.Objects;

public record IngestedRecord(
        GenericRecord record,
        String topic,

package com.nimbo.ingester.service;

import com.nimbo.ingester.config.KafkaProperties;
import com.nimbo.ingester.exception.DlqPublishException;

import org.apache.kafka.clients.consumer.ConsumerRecord;
import org.apache.kafka.clients.producer.KafkaProducer;
import org.apache.kafka.clients.producer.ProducerRecord;

import org.springframework.stereotype.Service;

import java.nio.charset.StandardCharsets;
import java.util.concurrent.ExecutionException;

@Service
public class DlqService {

    private final KafkaProducer<byte[], byte[]> kafkaProducer;
    private final KafkaProperties kafkaProperties;

    public DlqService(
            KafkaProducer<byte[], byte[]> kafkaProducer,
            KafkaProperties kafkaProperties
    ) {
        this.kafkaProducer = kafkaProducer;
        this.kafkaProperties = kafkaProperties;
    }


    public void publish(
            ConsumerRecord<byte[], byte[]> originalRecord,
            String errorType,
            String errorMessage
    ) {

        ProducerRecord<byte[], byte[]> dlqRecord =
                new ProducerRecord<>(kafkaProperties.getDlqTopic(), originalRecord.key(), originalRecord.value());


        addHeader(dlqRecord, "original-topic", originalRecord.topic());
        addHeader(dlqRecord, "original-partition", String.valueOf(originalRecord.partition()));
        addHeader(dlqRecord, "original-offset", String.valueOf(originalRecord.offset()));
        addHeader(dlqRecord, "error-type", errorType);
        addHeader(dlqRecord, "error-message", errorMessage);


        try {
            kafkaProducer.send(dlqRecord).get();

        } catch (InterruptedException e) {

            Thread.currentThread().interrupt();

            throw new DlqPublishException("Interrupted while publishing record to DLQ", e);

        } catch (ExecutionException e) {

            throw new DlqPublishException("Failed to publish record to DLQ", e);
        }
    }


    private void addHeader(
            ProducerRecord<byte[], byte[]> dlqRecord,
            String key,
            String value
    ) {

        if (value == null) {
            return;
        }

        dlqRecord.headers().add(key, value.getBytes(StandardCharsets.UTF_8));
    }
}

package com.nimbo.ingester.service;

import com.nimbo.ingester.model.ValidationResult;
import org.apache.avro.generic.GenericRecord;
import org.springframework.stereotype.Service;

@Service
public class NetflowValidationService {

    private static final String NULL_ERROR = " is null";
    private static final String INVALID_IPV4_ERROR = " invalid IPv4: ";

    public ValidationResult validate(GenericRecord genericRecord) {

        StringBuilder errors = new StringBuilder();

        checkIpv4(genericRecord, "IPV4_SRC_ADDR", errors);
        checkIpv4(genericRecord, "IPV4_DST_ADDR", errors);

        checkRange(
                genericRecord,
                "L4_SRC_PORT",
                0,
                65535,
                errors
        );

        checkRange(
                genericRecord,
                "L4_DST_PORT",
                0,
                65535,
                errors
        );

        checkNonNegative(
                genericRecord,
                "IN_BYTES",
                errors
        );

        checkNonNegative(
                genericRecord,
                "OUT_BYTES",
                errors
        );

        checkNonNegative(
                genericRecord,
                "IN_PKTS",
                errors
        );

        checkNonNegative(
                genericRecord,
                "OUT_PKTS",
                errors
        );

        checkRange(
                genericRecord,
                "MIN_TTL",
                0,
                255,
                errors
        );

        checkRange(
                genericRecord,
                "MAX_TTL",
                0,
                255,
                errors
        );

        checkMinMax(
                genericRecord,
                "MIN_TTL",
                "MAX_TTL",
                errors
        );

        checkNonNegative(
                genericRecord,
                "LONGEST_FLOW_PKT",
                errors
        );

        checkNonNegative(
                genericRecord,
                "SHORTEST_FLOW_PKT",
                errors
        );

        checkMinMax(
                genericRecord,
                "SHORTEST_FLOW_PKT",
                "LONGEST_FLOW_PKT",
                errors
        );

        checkNonNegative(
                genericRecord,
                "MIN_IP_PKT_LEN",
                errors
        );

        checkNonNegative(
                genericRecord,
                "MAX_IP_PKT_LEN",
                errors
        );

        checkMinMax(
                genericRecord,
                "MIN_IP_PKT_LEN",
                "MAX_IP_PKT_LEN",
                errors
        );

        if (errors.isEmpty()) {
            return ValidationResult.ok();
        }

        return ValidationResult.invalid(
                errors.toString()
        );
    }

    private void checkNonNegative(
            GenericRecord genericRecord,
            String field,
            StringBuilder errors
    ) {

        Object value = genericRecord.get(field);

        if (value == null) {
            addError(
                    errors,
                    field + NULL_ERROR
            );
            return;
        }

        long number =
                ((Number) value).longValue();

        if (number < 0) {
            addError(
                    errors,
                    field
                            + " is negative: "
                            + number
            );
        }
    }

    private void checkRange(
            GenericRecord genericRecord,
            String field,
            long min,
            long max,
            StringBuilder errors
    ) {

        Object value = genericRecord.get(field);

        if (value == null) {
            addError(
                    errors,
                    field + NULL_ERROR
            );
            return;
        }

        long number =
                ((Number) value).longValue();

        if (number < min || number > max) {
            addError(
                    errors,
                    field
                            + " out of range: "
                            + number
            );
        }
    }

    private void checkMinMax(
            GenericRecord genericRecord,
            String minField,
            String maxField,
            StringBuilder errors
    ) {

        Object minObject =
                genericRecord.get(minField);

        Object maxObject =
                genericRecord.get(maxField);

        if (minObject == null || maxObject == null) {
            return;
        }

        long min =
                ((Number) minObject).longValue();

        long max =
                ((Number) maxObject).longValue();

        if (min > max) {
            addError(
                    errors,
                    minField
                            + " > "
                            + maxField
                            + " ("
                            + min
                            + " > "
                            + max
                            + ")"
            );
        }
    }

    private void checkIpv4(
            GenericRecord genericRecord,
            String field,
            StringBuilder errors
    ) {

        Object value = genericRecord.get(field);

        if (value == null) {
            addError(
                    errors,
                    field + NULL_ERROR
            );
            return;
        }

        String ip =
                value.toString();

        String[] parts =
                ip.split("\\.", -1);

        if (parts.length != 4) {
            addError(
                    errors,
                    field
                            + INVALID_IPV4_ERROR
                            + ip
            );
            return;
        }

        try {
            for (String part : parts) {

                if (part.isEmpty()) {
                    throw new NumberFormatException();
                }

                int number =
                        Integer.parseInt(part);

                if (number < 0 || number > 255) {
                    addError(
                            errors,
                            field
                                    + INVALID_IPV4_ERROR
                                    + ip
                    );
                    return;
                }
            }

        } catch (NumberFormatException e) {

            addError(
                    errors,
                    field
                            + INVALID_IPV4_ERROR
                            + ip
            );
        }
    }

    private void addError(
            StringBuilder errors,
            String error
    ) {

        if (!errors.isEmpty()) {
            errors.append(" | ");
        }

        errors.append(error);
    }
}

package com.nimbo.metrics;

import com.nimbo.storage.service.batching.BatchPolicy;
import io.micrometer.core.instrument.Counter;
import io.micrometer.core.instrument.DistributionSummary;
import io.micrometer.core.instrument.Gauge;
import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Tags;
import io.micrometer.core.instrument.Timer;
import io.micrometer.core.instrument.binder.kafka.KafkaClientMetrics;
import lombok.extern.slf4j.Slf4j;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;
import java.util.function.ToDoubleFunction;

@Slf4j
@Component
public class BatchIngesterMetrics {

    private final MeterRegistry registry;
    private final Map<Integer, InstanceState> instances = new ConcurrentHashMap<>();
    private static final String INSTANCE_TAG = "instance";
    private static final String REASON_TAG = "reason";
    private static final List<String> ROLLOVER_REASONS = List.of("record_count", "size", "age");
    private static final List<String> DLQ_REASONS = List.of("VALIDATION_ERROR", "DESERIALIZATION_ERROR");

    public BatchIngesterMetrics(MeterRegistry registry) {
        this.registry = registry;
    }

    public void registerInstance(int instanceId) {
        instances.computeIfAbsent(instanceId, id -> {
            InstanceState s = new InstanceState(registry, id);
            String instance = String.valueOf(id);

            for (String reason : ROLLOVER_REASONS) {
                s.rolloverCounters.put(reason, registry.counter("ingester_rollover_total",
                        INSTANCE_TAG, instance, REASON_TAG, reason));
            }
            for (String reason : DLQ_REASONS) {
                s.dlqCounters.put(reason, registry.counter("ingester_dlq_records_total",
                        INSTANCE_TAG, instance, REASON_TAG, reason));
            }
            return s;
        });
    }

    public void bindKafkaConsumer(KafkaConsumer<?, ?> consumer) {
        new KafkaClientMetrics(consumer).bindTo(registry);
    }

    public <T> void registerGauge(String name, int instanceId, T target, ToDoubleFunction<T> valueFn) {
        Gauge.builder(name, target, valueFn)
                .tags(Tags.of(INSTANCE_TAG, String.valueOf(instanceId)))
                .register(registry);
    }

    private InstanceState state(int instanceId) {
        InstanceState s = instances.get(instanceId);
        if (s == null) {
            throw new IllegalStateException("Instance " + instanceId +
                    " not registered — call registerInstance() first");
        }
        return s;
    }

    public void consumerStarted(int instanceId) { state(instanceId).running.set(1); }
    public void consumerStopped(int instanceId) { state(instanceId).running.set(0); }
    public void recordsPolled(int instanceId, int count) { state(instanceId).pollRecords.record(count); }
    public void recordWritten(int instanceId) { state(instanceId).recordsWritten.increment(); }
    public void writeFailure(int instanceId) { state(instanceId).writeFailures.increment(); }

    public void invalidRecord(int instanceId, String reason) {
        InstanceState s = state(instanceId);
        Counter c = s.dlqCounters.get(reason);
        if (c != null) {
            c.increment();
        } else {
            log.warn("Unrecognized DLQ reason '{}' for instance {} — falling back to dynamic counter lookup",
                    reason, instanceId);

            registry.counter("ingester_dlq_records_total",
                    INSTANCE_TAG, String.valueOf(instanceId), REASON_TAG, reason).increment();
        }
    }

    public void rollover(int instanceId, BatchPolicy.FlushReason reason, long recordCount, long sizeBytes) {
        InstanceState s = state(instanceId);
        String reasonKey = reason.name().toLowerCase();
        Counter c = s.rolloverCounters.get(reasonKey);
        if (c != null) {
            c.increment();
        } else {
            log.warn("Unrecognized rollover reason '{}' for instance {} — falling back to dynamic counter lookup",
                    reasonKey, instanceId);

            registry.counter("ingester_rollover_total",
                    INSTANCE_TAG, String.valueOf(instanceId), REASON_TAG, reasonKey).increment();
        }
        s.batchRecordCount.record(recordCount);
        s.batchSizeBytes.record(sizeBytes);
        s.lastBatchRecordCount.set(recordCount);
        s.lastBatchSizeBytes.set(sizeBytes);
        s.lastSuccessfulRollover.set(System.currentTimeMillis() / 1000);
    }

    public Timer.Sample startRolloverTimer() { return Timer.start(registry); }

    public void stopRolloverTimer(int instanceId, Timer.Sample sample) {
        sample.stop(state(instanceId).rolloverDuration);
    }

    private static class InstanceState {
        final Counter recordsWritten;
        final Counter writeFailures;
        final Timer rolloverDuration;
        final DistributionSummary pollRecords;
        final DistributionSummary batchRecordCount;
        final DistributionSummary batchSizeBytes;
        final AtomicInteger running = new AtomicInteger(0);
        final AtomicLong lastSuccessfulRollover = new AtomicLong(0);
        final AtomicLong lastBatchRecordCount = new AtomicLong(0);
        final AtomicLong lastBatchSizeBytes = new AtomicLong(0);
        final Map<String, Counter> rolloverCounters = new ConcurrentHashMap<>();
        final Map<String, Counter> dlqCounters = new ConcurrentHashMap<>();

        InstanceState(MeterRegistry registry, int instanceId) {
            Tags tags = Tags.of(INSTANCE_TAG, String.valueOf(instanceId));

            recordsWritten = Counter.builder("ingester_records_written_total")
                    .tags(tags).register(registry);

            writeFailures = Counter.builder("ingester_write_failures_total")
                    .tags(tags).register(registry);

            rolloverDuration = Timer.builder("ingester_rollover_duration_seconds")
                    .tags(tags).register(registry);

            pollRecords = DistributionSummary.builder("ingester_poll_records")
                    .tags(tags).register(registry);

            batchRecordCount = DistributionSummary.builder("ingester_batch_record_count")
                    .tags(tags).register(registry);

            batchSizeBytes = DistributionSummary.builder("ingester_batch_size_bytes")
                    .tags(tags).register(registry);

            Gauge.builder("ingester_consumer_running", running, AtomicInteger::get)
                    .tags(tags).register(registry);

            Gauge.builder("ingester_last_successful_rollover_timestamp", lastSuccessfulRollover, AtomicLong::get)
                    .tags(tags).register(registry);

            Gauge.builder("ingester_last_batch_record_count", lastBatchRecordCount, AtomicLong::get)
                    .tags(tags).register(registry);

            Gauge.builder("ingester_last_batch_size_bytes", lastBatchSizeBytes, AtomicLong::get)
                    .tags(tags).register(registry);
        }
    }
}

package com.nimbo.storage.service;

import com.nimbo.ingester.model.IngestedRecord;
import com.nimbo.metrics.BatchIngesterMetrics;
import com.nimbo.storage.config.HdfsProperties;
import com.nimbo.storage.exception.ParquetWriteException;
import com.nimbo.storage.exception.StorageException;
import com.nimbo.storage.model.TopicPartitionKey;
import com.nimbo.storage.service.batching.BatchAccumulator;
import com.nimbo.storage.service.batching.BatchPolicy;
import com.nimbo.storage.service.hdfs.FileStorage;
import com.nimbo.storage.service.naming.FileNameGenerator;
import com.nimbo.storage.service.parquet.ParquetFileWriter;
import io.micrometer.core.instrument.Timer;
import lombok.extern.slf4j.Slf4j;
import org.apache.avro.Schema;
import org.apache.avro.generic.GenericData;
import org.apache.avro.generic.GenericRecord;
import org.apache.hadoop.fs.Path;
import org.apache.parquet.hadoop.ParquetWriter;

import java.time.Duration;
import java.time.Instant;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;

@Slf4j
public final class NetFlowStorage {

    private final ParquetFileWriter parquetFileWriter;
    private final FileStorage fileStorage;
    private final FileNameGenerator fileNameGenerator;
    private final HdfsProperties hdfsProperties;
    private final Schema storageSchema;

    private final BatchAccumulator batchAccumulator;

    private ParquetWriter<GenericRecord> currentWriter;
    private Path currentTempPath;
    private Path currentFinalPath;
    private boolean writerOpen = false;

    private final BatchIngesterMetrics metrics;
    private final int instanceId;

    public NetFlowStorage(
            BatchPolicy batchPolicy,
            ParquetFileWriter parquetFileWriter,
            FileStorage fileStorage,
            FileNameGenerator fileNameGenerator,
            HdfsProperties hdfsProperties,
            Schema storageSchema,
            BatchIngesterMetrics metrics,
            int instanceId
    ) {
        this.parquetFileWriter = Objects.requireNonNull(parquetFileWriter);
        this.fileStorage = Objects.requireNonNull(fileStorage);
        this.fileNameGenerator = Objects.requireNonNull(fileNameGenerator);
        this.hdfsProperties = Objects.requireNonNull(hdfsProperties);
        this.storageSchema = Objects.requireNonNull(storageSchema);
        this.batchAccumulator = new BatchAccumulator(batchPolicy);
        this.metrics = metrics;
        this.instanceId = instanceId;

        metrics.registerGauge("ingester_open_batch_record_count", instanceId,
                batchAccumulator, BatchAccumulator::getRecordCount);
        metrics.registerGauge("ingester_open_batch_age_seconds", instanceId,
                this, NetFlowStorage::currentBatchAgeSeconds);

        log.info("NetFlowStorage initialized (streaming writer) with HDFS target path: {}", hdfsProperties.getBasePath());
    }

    private double currentBatchAgeSeconds() {
        if (!writerOpen) return 0;
        Instant firstRecordAt = batchAccumulator.getFirstRecordAt();
        return firstRecordAt == null ? 0 : Duration.between(firstRecordAt, Instant.now()).getSeconds();
    }

    public Optional<Map<TopicPartitionKey, Long>> accept(IngestedRecord ingestedRecord) {
        Objects.requireNonNull(ingestedRecord, "ingestedRecord must not be null");

        try {
            if (!writerOpen) {
                openWriter(ingestedRecord);
            }

            GenericRecord enrichedRecord = enrichWithProcessingTime(ingestedRecord.record());
            parquetFileWriter.write(currentWriter, enrichedRecord);
            metrics.recordWritten(instanceId);
            batchAccumulator.add(ingestedRecord);

            if (batchAccumulator.shouldFlush()) {
                Map<TopicPartitionKey, Long> offsets = commitCurrentBatch();
                return Optional.of(offsets);
            }

            return Optional.empty();

        } catch (StorageException e) {
            handleWriteFailure();

            log.error(
                    "Failed writing record: topic={}, partition={}, offset={}",
                    ingestedRecord.topic(),
                    ingestedRecord.partition(),
                    ingestedRecord.offset(),
                    e
            );

            throw e;
        }
    }

    public Map<TopicPartitionKey, Long> flush() {
        if (batchAccumulator.isEmpty()) {
            log.debug("Flush requested but accumulator is empty");
            return Map.of();
        }

        log.info("Executing lifecycle flush for {} records", batchAccumulator.getRecordCount());

        try {
            return commitCurrentBatch();
        } catch (StorageException e) {
            handleWriteFailure();
            throw e;
        }
    }

    public Map<TopicPartitionKey, Long> flushRemaining() {
        if (batchAccumulator.isEmpty()) {
            return Map.of();
        }

        if (!batchAccumulator.shouldFlush()) {
            return Map.of();
        }

        log.info(
                "Flushed remaining records. Record count: {}, accumulated size: {} bytes",
                batchAccumulator.getRecordCount(),
                batchAccumulator.getAccumulatedSizeBytes()
        );

        try {
            return commitCurrentBatch();
        } catch (StorageException e) {
            handleWriteFailure();
            throw e;
        }
    }

    public Optional<Map<TopicPartitionKey, Long>> recordSkippedOffset(String topic, int partition, long nextOffset) {
        return batchAccumulator.recordSkippedOffset(topic, partition, nextOffset);
    }

    private void openWriter(IngestedRecord firstRecord)  {
        int primaryPartition = firstRecord.partition();

        String finalFileName = fileNameGenerator.generateFileName(primaryPartition);
        String tempFileName = fileNameGenerator.generateTempFileName(finalFileName);

        Path basePath = new Path(hdfsProperties.getBasePath());
        currentFinalPath = new Path(basePath, finalFileName);
        currentTempPath = new Path(basePath, tempFileName);

        fileStorage.ensureDirectoryExists(currentFinalPath.getParent());
        currentWriter = parquetFileWriter.createWriter(currentTempPath, storageSchema);
        writerOpen = true;

        log.debug("Opened streaming writer for partition {} at {}", primaryPartition, currentTempPath);
    }

    private Map<TopicPartitionKey, Long> commitCurrentBatch() {
        if (!writerOpen) {
            return Map.of();
        }

        long recordCount = batchAccumulator.getRecordCount();
        long sizeBytes = batchAccumulator.getAccumulatedSizeBytes();
        BatchPolicy.FlushReason reason = batchAccumulator.flushReason();

        Timer.Sample sample = metrics.startRolloverTimer();
        parquetFileWriter.closeWriter(currentWriter);
        writerOpen = false;

        fileStorage.commitAtomic(currentTempPath, currentFinalPath);

        metrics.stopRolloverTimer(instanceId, sample);
        metrics.rollover(instanceId, reason, recordCount, sizeBytes);

        Map<TopicPartitionKey, Long> offsets = batchAccumulator.getPendingOffsets();

        batchAccumulator.clear();

        currentWriter = null;
        currentTempPath = null;
        currentFinalPath = null;

        return offsets;
    }

    private void handleWriteFailure() {
        metrics.writeFailure(instanceId);
        if (writerOpen) {
            try {
                parquetFileWriter.closeWriter(currentWriter);
            } catch (ParquetWriteException closeException) {
                log.warn("Failed closing writer during error handling", closeException);
            }
            writerOpen = false;
        }

        if (currentTempPath != null) {
            fileStorage.cleanupTempFile(currentTempPath);
        }

        batchAccumulator.clear();

        currentWriter = null;
        currentTempPath = null;
        currentFinalPath = null;
    }

    private GenericRecord enrichWithProcessingTime(GenericRecord original) {
        GenericRecord enriched = new GenericData.Record(storageSchema);

        Instant now = Instant.now();
        long processingTimeMicros = now.getEpochSecond() * 1_000_000 + now.getNano() / 1000;

        for (Schema.Field field : storageSchema.getFields()) {
            String fieldName = field.name();
            if ("PROCESSING_TIME".equals(fieldName)) {
                enriched.put("PROCESSING_TIME", processingTimeMicros);
            } else {
                enriched.put(fieldName, original.get(fieldName));
            }
        }
        return enriched;
    }
}

package com.nimbo.storage.service.batching;

import com.nimbo.ingester.model.IngestedRecord;
import com.nimbo.storage.model.TopicPartitionKey;
import lombok.Getter;
import lombok.extern.slf4j.Slf4j;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.time.Instant;
import java.util.HashMap;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;

@Slf4j
public final class BatchAccumulator {

    private final BatchPolicy batchPolicy;

    @Getter
    private Instant firstRecordAt;
    @Getter
    private long accumulatedSizeBytes;
    @Getter
    private long recordCount;

    // Nested map: topic -> partition -> nextOffset
    private final Map<String, Map<Integer, Long>> pendingOffsets = new HashMap<>();

    public BatchAccumulator(BatchPolicy batchPolicy) {
        this.batchPolicy = Objects.requireNonNull(
                batchPolicy,
                "batchPolicy must not be null"
        );
    }

    public void add(IngestedRecord ingestedRecord) {
        Objects.requireNonNull(ingestedRecord, "record must not be null");

        if (firstRecordAt == null) {
            firstRecordAt = Instant.now();
        }

        recordCount++;
        accumulatedSizeBytes += ingestedRecord.serializedSizeBytes();

        pendingOffsets
                .computeIfAbsent(ingestedRecord.topic(), t -> new HashMap<>())
                .merge(ingestedRecord.partition(), ingestedRecord.offset() + 1, Math::max);
    }

    public boolean shouldFlush() {
        boolean flush = batchPolicy.shouldFlush(
                recordCount,
                accumulatedSizeBytes,
                firstRecordAt,
                Instant.now()
        );

        if (flush) {
            log.info(
                    "Batch flush triggered: recordCount={}, accumulatedSizeBytes={}, firstRecordAt={}",
                    recordCount,
                    accumulatedSizeBytes,
                    firstRecordAt
            );
        }

        return flush;
    }

    public BatchPolicy.FlushReason flushReason() {
        return batchPolicy.flushReason(recordCount, accumulatedSizeBytes, firstRecordAt, Instant.now());
    }

    public void clear() {
        recordCount = 0;
        accumulatedSizeBytes = 0L;
        firstRecordAt = null;
        pendingOffsets.clear();
    }

    public Optional<Map<TopicPartitionKey, Long>> recordSkippedOffset(String topic, int partition, long nextOffset) {

        if (firstRecordAt == null) {
            firstRecordAt = Instant.now();
        }

        pendingOffsets
                .computeIfAbsent(topic, t -> new HashMap<>())
                .merge(partition, nextOffset, Math::max);

        return Optional.empty();
    }

    public Map<TopicPartitionKey, Long> getPendingOffsets() {
        Map<TopicPartitionKey, Long> result = new HashMap<>();
        pendingOffsets.forEach((topic, partitions) ->
                partitions.forEach((partition, offset) ->
                        result.put(
                                new TopicPartitionKey(topic, partition),
                                offset
                        )
                )
        );
        return Map.copyOf(result);
    }

    public boolean isEmpty() {
        return recordCount == 0;
    }
}

package com.nimbo.storage.service.batching;

import java.time.Duration;
import java.time.Instant;
import java.util.Objects;

public record BatchPolicy(long maxRecordCount, long maxSizeBytes, Duration maxBatchAge) {

    public BatchPolicy(
            long maxRecordCount,
            long maxSizeBytes,
            Duration maxBatchAge
    ) {
        if (maxRecordCount <= 0) {
            throw new IllegalArgumentException("maxRecordCount must be positive");
        }

        if (maxSizeBytes <= 0) {
            throw new IllegalArgumentException("maxSizeBytes must be positive");
        }

        this.maxBatchAge = Objects.requireNonNull(maxBatchAge, "maxBatchAge must not be null");

        if (maxBatchAge.isZero() || maxBatchAge.isNegative()) {
            throw new IllegalArgumentException("maxBatchAge must be positive");
        }

        this.maxRecordCount = maxRecordCount;
        this.maxSizeBytes = maxSizeBytes;
    }

    public enum FlushReason { RECORD_COUNT, SIZE, AGE, NONE }

    public boolean shouldFlush(
            long recordCount,
            long totalSizeBytes,
            Instant firstRecordAt,
            Instant now
    ) {
        Objects.requireNonNull(now, "now must not be null");

        if (recordCount == 0) {
            return false;
        }

        if (recordCount >= maxRecordCount) {
            return true;
        }

        if (totalSizeBytes >= maxSizeBytes) {
            return true;
        }

        if (firstRecordAt == null) {
            return false;
        }

        return Duration.between(firstRecordAt, now)
                .compareTo(maxBatchAge) >= 0;
    }

    public FlushReason flushReason(long recordCount, long totalSizeBytes, Instant firstRecordAt, Instant now) {
        if (recordCount == 0) {
            return FlushReason.NONE;
        }
        if (recordCount >= maxRecordCount) {
            return FlushReason.RECORD_COUNT;
        }
        if (totalSizeBytes >= maxSizeBytes) {
            return FlushReason.SIZE;
        }
        if (firstRecordAt != null && Duration.between(firstRecordAt, now).compareTo(maxBatchAge) >= 0) {
            return FlushReason.AGE;
        }
        return FlushReason.NONE;
    }
}

package com.nimbo.storage.service.hdfs;

import com.nimbo.storage.exception.HdfsStorageException;
import lombok.extern.slf4j.Slf4j;
import org.apache.hadoop.fs.FileSystem;
import org.apache.hadoop.fs.Path;

import java.io.IOException;
import java.util.Objects;

@Slf4j
public final class HdfsFileStorage implements FileStorage {

    private final FileSystem fileSystem;

    public HdfsFileStorage(FileSystem fileSystem) {
        this.fileSystem = Objects.requireNonNull(fileSystem, "fileSystem must not be null");
    }

    @Override
    public void ensureDirectoryExists(Path path) {
        Objects.requireNonNull(path, "path must not be null");

        try {
            if (fileSystem.exists(path)) {
                return;
            }
            boolean created = fileSystem.mkdirs(path);
            if (created) {
                log.info("Created HDFS directory: {}", path);
                return;
            }
            if (!fileSystem.exists(path)) {
                throw new HdfsStorageException(
                        "Failed to create HDFS directory: " + path
                );
            }
            log.debug("HDFS directory already created by another consumer: {}", path);

        } catch (IOException e) {
            throw new HdfsStorageException(
                    "Failed accessing HDFS directory: " + path,
                    e
            );
        }
    }

    @Override
    public void commitAtomic(Path tempPath, Path finalPath) {
        Objects.requireNonNull(tempPath, "tempPath must not be null");
        Objects.requireNonNull(finalPath, "finalPath must not be null");

        try {
            if (fileSystem.exists(finalPath)) {
                throw new HdfsStorageException(
                        "Target Parquet file already exists: " + finalPath
                );
            }
            if (!fileSystem.rename(tempPath, finalPath)) {
                throw new HdfsStorageException(
                        "Failed to rename temporary Parquet file: "
                                + tempPath + " -> " + finalPath
                );
            }
        } catch (IOException e) {
            throw new HdfsStorageException(
                    "Failed committing temporary Parquet file: "
                            + tempPath + " -> " + finalPath, e
            );
        }
    }

    @Override
    public void cleanupTempFile(Path tempPath) {
        Objects.requireNonNull(tempPath, "tempPath must not be null");

        try {
            if (fileSystem.exists(tempPath)) {
                boolean deleted = fileSystem.delete(tempPath, false);
                if (!deleted) {
                    log.warn("Failed to delete temporary HDFS file: {}", tempPath);
                }
            }
        } catch (IOException e) {
            log.warn("Failed during temporary file cleanup: {}", tempPath, e);
        }
    }
}

package com.nimbo.storage.service.hdfs;

import com.nimbo.storage.config.HdfsProperties;
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.FileSystem;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.net.URI;
import java.util.Map;

@Component
public class HdfsFileSystemFactory {


    public FileSystem create(HdfsProperties hdfsProperties)
            throws IOException, InterruptedException {
        Configuration conf = new Configuration();
        String uriStr = hdfsProperties.getUri();
        conf.set("fs.defaultFS", uriStr);
        conf.set("fs.hdfs.impl", org.apache.hadoop.hdfs.DistributedFileSystem.class.getName());
        conf.set("fs.file.impl", org.apache.hadoop.fs.LocalFileSystem.class.getName());
        conf.set("dfs.replication", "3");

        String nameservice = hdfsProperties.getNameservice();
        if (nameservice != null && !nameservice.isBlank()) {

            conf.set("dfs.nameservices", nameservice);
            Map<String,String> nodes = hdfsProperties.getNodes();
            if (nodes != null && !nodes.isEmpty()) {

                conf.set(
                        "dfs.ha.namenodes." + nameservice,
                        String.join(",", nodes.keySet())
                );

                conf.setBoolean(
                        "dfs.client.failover.random.order",
                        false
                );

                for (Map.Entry<String,String> entry : nodes.entrySet()) {

                    conf.set(
                            "dfs.namenode.rpc-address."
                                    + nameservice
                                    + "."
                                    + entry.getKey(),
                            entry.getValue()
                    );
                }
            }

            conf.set(
                    "dfs.client.failover.proxy.provider." + nameservice,
                    "org.apache.hadoop.hdfs.server.namenode.ha.ConfiguredFailoverProxyProvider"
            );
        }

        return FileSystem.newInstance(URI.create(uriStr), conf,"hadoop");
    }
}

package com.nimbo.storage.service.naming;

import org.springframework.stereotype.Component;

import java.time.Clock;
import java.time.Instant;
import java.time.ZoneOffset;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import java.util.UUID;
import java.util.function.Supplier;

@Component
public final class FileNameGenerator {

    private static final DateTimeFormatter TIMESTAMP_FORMATTER =
            DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH-mm-ss-SSS'Z'").withZone(ZoneOffset.UTC);
    private static final DateTimeFormatter DATE_FORMATTER =
            DateTimeFormatter.ofPattern("yyyy-MM-dd").withZone(ZoneOffset.UTC);
    private static final DateTimeFormatter HOUR_FORMATTER =
            DateTimeFormatter.ofPattern("HH").withZone(ZoneOffset.UTC);

    private final Clock clock;
    private final Supplier<UUID> uuidSupplier;

    public FileNameGenerator() {
        this(Clock.systemUTC(), UUID::randomUUID);
    }

    FileNameGenerator(Clock clock, Supplier<UUID> uuidSupplier) {
        this.clock = Objects.requireNonNull(clock, "clock must not be null");
        this.uuidSupplier = Objects.requireNonNull(uuidSupplier, "uuidSupplier must not be null");
    }

    public String generateFileName(int partition) {
        if (partition < 0) {
            throw new IllegalArgumentException("partition must not be negative");
        }
        Instant openedAt = clock.instant();
        String fileName = String.format(
                "netflow-p%02d-%s-%s.parquet",
                partition,
                TIMESTAMP_FORMATTER.format(openedAt),
                uuidSupplier.get()
        );
        return String.join(
                "/",
                DATE_FORMATTER.format(openedAt),
                HOUR_FORMATTER.format(openedAt),
                fileName
        );
    }

    public String generateTempFileName(String finalFileName) {
        Objects.requireNonNull(finalFileName, "finalFileName must not be null");

        int separator = finalFileName.lastIndexOf('/');
        String parent = separator >= 0 ? finalFileName.substring(0, separator + 1) : "";
        String fileName = finalFileName.substring(separator + 1);
        return parent + "." + fileName + ".tmp";
    }
}

package com.nimbo.storage.service.parquet;

import com.nimbo.storage.config.ParquetProperties;
import com.nimbo.storage.exception.ParquetWriteException;
import org.apache.avro.Schema;
import org.apache.avro.generic.GenericRecord;
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.FileSystem;
import org.apache.hadoop.fs.Path;
import org.apache.parquet.avro.AvroParquetWriter;
import org.apache.parquet.hadoop.ParquetWriter;
import org.apache.parquet.hadoop.metadata.CompressionCodecName;

import java.io.IOException;
import java.util.Objects;

public final class ParquetFileWriter {

    private final Configuration hadoopConfiguration;
    private final ParquetProperties parquetProperties;

    public ParquetFileWriter(FileSystem fileSystem, ParquetProperties parquetProperties) {
        Objects.requireNonNull(fileSystem, "fileSystem must not be null");
        this.parquetProperties = Objects.requireNonNull(parquetProperties, "parquetProperties must not be null");
        this.hadoopConfiguration = fileSystem.getConf();
    }

    public ParquetWriter<GenericRecord> createWriter(Path targetPath, Schema schema)
            throws ParquetWriteException {
        Objects.requireNonNull(targetPath, "targetPath must not be null");
        Objects.requireNonNull(schema, "schema must not be null");

        CompressionCodecName codec = CompressionCodecName.valueOf(
                parquetProperties.getCompressionCodec().toUpperCase()
        );

        return buildWriter(targetPath, schema, codec);
    }

    public void write(ParquetWriter<GenericRecord> writer, GenericRecord ingestedRecord)
            throws ParquetWriteException {
        Objects.requireNonNull(writer, "writer must not be null");
        Objects.requireNonNull(ingestedRecord, "record must not be null");

        try {
            writer.write(ingestedRecord);
        } catch (IOException e) {
            throw new ParquetWriteException("Failed writing record to Parquet writer", e);
        }
    }

    public void closeWriter(ParquetWriter<GenericRecord> writer)
            throws ParquetWriteException {
        Objects.requireNonNull(writer, "writer must not be null");

        try {
            writer.close();
        } catch (IOException e) {
            throw new ParquetWriteException("Failed closing Parquet writer", e);
        }
    }

    private ParquetWriter<GenericRecord> buildWriter(
            Path targetPath,
            Schema schema,
            CompressionCodecName codec
    ) throws ParquetWriteException {
        try {
            return AvroParquetWriter
                    .<GenericRecord>builder(targetPath)
                    .withSchema(schema)
                    .withConf(hadoopConfiguration)
                    .withCompressionCodec(codec)
                    .withRowGroupSize(parquetProperties.getBlockSizeBytes())
                    .withPageSize(parquetProperties.getPageSizeBytes())
                    .withDictionaryEncoding(parquetProperties.isEnableDictionary())
                    .build();
        } catch (IOException e) {
            throw new ParquetWriteException("Failed creating Parquet writer for path: " + targetPath, e);
        }
    }
}

                "type": "record",
                "name": "TestRecord",
                "fields": [
                    {"name": "id", "type": "string"},
                    {"name": "value", "type": "long"}
                ]
            }
            """;

    private Schema schema;
    private AvroDeserializer deserializer;

    @BeforeEach
    void setUp() {
        schema = new Schema.Parser().parse(SCHEMA_JSON);
        deserializer = new AvroDeserializer(schema);
    }

    @Test
    @DisplayName("Should deserialize valid Avro data successfully")
    void shouldDeserializeValidAvroData() throws IOException {
        GenericRecord expected = new GenericData.Record(schema);
        expected.put("id", "test-123");
        expected.put("value", 42L);

        byte[] data = serializeRecord(expected);

        GenericRecord actual = deserializer.deserialize(data);

        assertThat(actual).isNotNull();
        assertThat(actual.get("id").toString()).isEqualTo("test-123");
        assertThat(actual.get("value")).isEqualTo(42L);
    }

    @Test
    @DisplayName("Should throw DeserializationException when data is null")
    void shouldThrowExceptionWhenDataIsNull() {
        assertThatThrownBy(() -> deserializer.deserialize(null))
                .isInstanceOf(DeserializationException.class);
    }

    @Test
    @DisplayName("Should return correct schema")
    void shouldReturnCorrectSchema() {
        assertThat(deserializer.getSchema()).isSameAs(schema);
        assertThat(deserializer.getSchema().getFullName()).isEqualTo("TestRecord");
    }

    private byte[] serializeRecord(GenericRecord record) throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Encoder encoder = EncoderFactory.get().binaryEncoder(out, null);
        DatumWriter<GenericRecord> writer = new GenericDatumWriter<>(schema);
        writer.write(record, encoder);
        encoder.flush();
        return out.toByteArray();
    }
}

    private void awaitCondition(java.util.function.BooleanSupplier condition, Duration timeout, String failMessage)
            throws InterruptedException {
        long deadline = System.currentTimeMillis() + timeout.toMillis();
        while (System.currentTimeMillis() < deadline) {
            if (condition.getAsBoolean()) {
                return;
            }
            Thread.sleep(200);
        }
        fail(failMessage);
    }

    private ConsumerRecord<byte[], byte[]> pollForOne(KafkaConsumer<byte[], byte[]> consumer, Duration timeout) {
        return await()
                .atMost(timeout)
                .pollInterval(Duration.ofMillis(300))
                .until(() -> {
                    ConsumerRecords<byte[], byte[]> records = consumer.poll(Duration.ofMillis(100));
                    return records.isEmpty() ? null : records.iterator().next();
                }, Objects::nonNull);
    }

    private List<FileStatus> finalizedParquetFiles(Path storageDirectory) throws IOException {
        if (!fileSystem.exists(storageDirectory)) {
            return List.of();
        }

        List<FileStatus> files = new ArrayList<>();
        RemoteIterator<LocatedFileStatus> iterator = fileSystem.listFiles(storageDirectory, true);
        while (iterator.hasNext()) {
            LocatedFileStatus file = iterator.next();
            String fileName = file.getPath().getName();
            if (!fileName.startsWith(".") && fileName.endsWith(".parquet")) {
                files.add(file);
            }
        }
        return files;
    }

    private String headerValue(ConsumerRecord<byte[], byte[]> record, String key) {
        var header = record.headers().lastHeader(key);
        return header == null ? null : new String(header.value(), java.nio.charset.StandardCharsets.UTF_8);
    }

    private void closeQuietly(AutoCloseable closeable) {
        if (closeable == null) return;
        try {
            closeable.close();
        } catch (Exception ignored) {
        }
    }

        assertEquals(3, offsets.size());
        assertEquals(11L, offsets.get(new TopicPartitionKey("netflow", 0)));
        assertEquals(21L, offsets.get(new TopicPartitionKey("netflow", 1)));
        assertEquals(31L, offsets.get(new TopicPartitionKey("netflow", 2)));
    }

    @Test

class BatchPolicyTest {

    private static final Instant BASE_TIME = Instant.parse("2026-08-11T10:00:00Z");

    private final BatchPolicy policy = new BatchPolicy(100, 1_000, Duration.ofMinutes(2));

    @Test
    void shouldNotFlushEmptyBatch() {
        boolean result = policy.shouldFlush(0, 0, null, BASE_TIME);
        assertFalse(result);
    }

    @Test
    void shouldFlushWhenRecordCountReachesLimit() {
        boolean result = policy.shouldFlush(100, 500, BASE_TIME, BASE_TIME);
        assertTrue(result);
    }

    @Test
    void shouldNotFlushWhenRecordCountIsBelowLimit() {
        boolean result = policy.shouldFlush(99, 500, BASE_TIME, BASE_TIME);
        assertFalse(result);
    }

    @Test
    void shouldFlushWhenSizeReachesLimit() {
        boolean result = policy.shouldFlush(50, 1_000, BASE_TIME, BASE_TIME);
        assertTrue(result);
    }

    @Test
    void shouldNotFlushWhenSizeIsBelowLimit() {
        boolean result = policy.shouldFlush(50, 999, BASE_TIME, BASE_TIME);
        assertFalse(result);
    }

    @Test
    void shouldFlushWhenBatchAgeReachesLimit() {
        Instant now = BASE_TIME.plus(Duration.ofMinutes(2));
        boolean result = policy.shouldFlush(50, 500, BASE_TIME, now);

        assertTrue(result);
    }

    @Test
    void shouldNotFlushWhenBatchAgeIsBelowLimit() {
        Instant now = BASE_TIME.plus(Duration.ofMinutes(2)).minusMillis(1);
        boolean result = policy.shouldFlush(50, 500, BASE_TIME, now);

        assertFalse(result);
    }

    void shouldNotFlushWhenFirstRecordTimeIsNull() {
        boolean result = policy.shouldFlush(50, 500, null, BASE_TIME);

        assertFalse(result);
    }

    }

    @Test
    void shouldRejectZeroMaxRecordCount() {
        assertThrows(IllegalArgumentException.class, () -> new BatchPolicy(0, 1_000, Duration.ofMinutes(2)));
    }

    @Test
    void shouldRejectNegativeMaxRecordCount() {
        assertThrows(IllegalArgumentException.class, () -> new BatchPolicy(-1, 1_000, Duration.ofMinutes(2)));
    }

    @Test
    void shouldRejectZeroMaxSize() {

    }

    @Test
    void shouldRejectNegativeMaxSize() {

    }

package com.nimbo.storage.service.naming;

import org.junit.jupiter.api.Test;

import java.time.Clock;
import java.time.Instant;
import java.time.ZoneOffset;
import java.util.UUID;

import static org.assertj.core.api.Assertions.assertThat;

class FileNameGeneratorTest {

    @Test
    void givenAFixedWriterOpeningTime_whenFileNameIsGenerated_thenDirectoryAndNameUseTheSameUtcHour() {
        // Given: a writer opening just before a UTC hour boundary
        UUID fileId = UUID.fromString("50000000-0000-0000-0000-000000000001");
        FileNameGenerator generator = new FileNameGenerator(
                Clock.fixed(Instant.parse("2026-09-07T13:59:59.987Z"), ZoneOffset.UTC),
                () -> fileId
        );

        // When: the partitioned relative file name is generated
        String fileName = generator.generateFileName(3);

        // Then: date, hour, filename timestamp, partition, and identity are consistent
        assertThat(fileName).isEqualTo(
                "2026-09-07/13/netflow-p03-2026-09-07T13-59-59-987Z-50000000-0000-0000-0000-000000000001.parquet"
        );
    }

    @Test
    void givenAPartitionedFinalName_whenTempNameIsGenerated_thenBothRemainInTheSameDirectory() {
        // Given: a committed filename inside an hourly partition
        FileNameGenerator generator = new FileNameGenerator();
        String finalName = "2026-09-07/14/netflow-p01-2026-09-07T14-00-00-000Z-file.parquet";

        // When: its hidden temporary name is derived
        String temporaryName = generator.generateTempFileName(finalName);

        // Then: only the basename changes and the hourly directory is preserved
        assertThat(temporaryName).isEqualTo(
                "2026-09-07/14/.netflow-p01-2026-09-07T14-00-00-000Z-file.parquet.tmp"
        );
    }

    @Test
    void givenAnInvalidKafkaPartition_whenFileNameIsGenerated_thenItIsRejected() {
        // Given: a generator and a negative Kafka partition
        FileNameGenerator generator = new FileNameGenerator();

        // When: a filename is requested for the invalid partition
        Throwable failure = org.assertj.core.api.Assertions.catchThrowable(() -> generator.generateFileName(-1));

        // Then: no undiscoverable filename can be produced
        assertThat(failure).isInstanceOf(IllegalArgumentException.class).hasMessageContaining("partition");
    }
}

package com.nimbo.storage.service.parquet;

import com.nimbo.storage.config.ParquetProperties;
import org.apache.avro.Schema;
import org.apache.avro.generic.GenericData;
import org.apache.avro.generic.GenericRecord;
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.FileSystem;
import org.apache.hadoop.fs.Path;
import org.apache.parquet.hadoop.ParquetWriter;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.io.TempDir;

import java.net.URI;

import static org.junit.jupiter.api.Assertions.*;

class ParquetFileWriterTest {

    @TempDir
    java.nio.file.Path tempDir;

    private ParquetProperties createProperties() {

        ParquetProperties properties = new ParquetProperties();
        properties.setCompressionCodec("SNAPPY");
        properties.setBlockSizeBytes(1024 * 1024);
        properties.setPageSizeBytes(1024 * 1024);
        properties.setEnableDictionary(true);

        return properties;
    }

    private Schema createSchema() {

        return new Schema.Parser()
                .parse("""
                        {
                          "type":"record",
                          "name":"Test",
                          "fields":[
                            {
                              "name":"value",
                              "type":"string"
                            }
                          ]
                        }
                        """
                );
    }

    private FileSystem createFileSystem() throws Exception {

        Configuration conf = new Configuration();
        conf.set("fs.defaultFS", "file:///");
        return FileSystem.newInstance(URI.create("file:///"), conf);
    }

    @Test
    void shouldCreateWriteAndCloseParquetWriter() throws Exception {

        FileSystem fs = createFileSystem();
        ParquetFileWriter writerService = new ParquetFileWriter(fs, createProperties());
        Path path = new Path(tempDir.resolve("test.parquet").toString());
        Schema schema = createSchema();
        ParquetWriter<GenericRecord> writer = writerService.createWriter(path, schema);
        GenericRecord testRecord = new GenericData.Record(schema);
        testRecord.put("value", "hello");
        writerService.write(writer, testRecord);
        writerService.closeWriter(writer);


        assertTrue(fs.exists(path));
    }

    @Test
    void shouldRejectNullWriterDependencies() throws Exception {

        FileSystem fs = createFileSystem();
        ParquetProperties properties = createProperties();


        assertThrows(NullPointerException.class, () -> new ParquetFileWriter(null, properties));
        assertThrows(NullPointerException.class, () -> new ParquetFileWriter(fs, null));
    }

    @Test
    void shouldRejectNullCreateWriterArguments() throws Exception {

        FileSystem fs = createFileSystem();
        ParquetProperties properties = createProperties();
        ParquetFileWriter writerService = new ParquetFileWriter(fs, properties);
        Path validPath = new Path("/tmp/test.parquet");
        Schema validSchema = createSchema();


        assertThrows(NullPointerException.class, () -> writerService.createWriter(null, validSchema));
        assertThrows(NullPointerException.class, () -> writerService.createWriter(validPath, null));
    }

    @Test
    void shouldRejectNullWriteArguments() throws Exception {

        FileSystem fs = createFileSystem();
        ParquetFileWriter writerService = new ParquetFileWriter(fs, createProperties());
        assertThrows(NullPointerException.class, () -> writerService.write(null, null));
    }
}
