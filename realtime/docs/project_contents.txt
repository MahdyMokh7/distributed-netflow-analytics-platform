package com.nimbo.realtime.avro;

import org.apache.avro.AvroRuntimeException;
import org.apache.avro.Schema;
import org.apache.avro.generic.GenericDatumReader;
import org.apache.avro.generic.GenericRecord;
import org.apache.avro.io.BinaryDecoder;
import org.apache.avro.io.DecoderFactory;
import org.springframework.stereotype.Component;

import java.io.IOException;

@Component
public class AvroDeserializer {

    private final Schema schema;

    public AvroDeserializer(Schema netflowSchema) {
        this.schema = netflowSchema;
    }

    public GenericRecord deserialize(byte[] data) {
        if (data == null || data.length == 0) {
            throw new AvroDeserializationException(
                    "Cannot deserialize empty Avro payload",
                    new IllegalArgumentException("Avro payload is null or empty")
            );
        }

        GenericDatumReader<GenericRecord> reader = new GenericDatumReader<>(schema);
        BinaryDecoder decoder = DecoderFactory.get().binaryDecoder(data, null);

        try {
            return reader.read(null, decoder);
        } catch (IOException | AvroRuntimeException e) {
            throw new AvroDeserializationException("Failed to deserialize NetFlow Avro record", e);
        }
    }
}

package com.nimbo.realtime.client;

import com.nimbo.realtime.config.StarRocksProperties;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.concurrent.atomic.AtomicInteger;

@Component
public class BackendSelector {

    private final List<String> backends;
    private final AtomicInteger index = new AtomicInteger();

    public BackendSelector(StarRocksProperties properties) {

        this.backends = List.copyOf(
                properties.getStreamLoadBackends()
        );

        if (backends.isEmpty()) {
            throw new IllegalStateException(
                    "At least one StarRocks backend must be configured"
            );
        }
    }

    public String nextBackend() {
        int position = Math.floorMod(
                index.getAndIncrement(),
                backends.size()
        );
        return backends.get(position);
    }
}

package com.nimbo.realtime.client;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.nimbo.realtime.config.StarRocksProperties;
import com.nimbo.realtime.mapper.SchemaMapping;
import com.nimbo.realtime.model.StarRocksRow;
import com.nimbo.realtime.model.StreamLoadResult;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.Base64;
import java.util.List;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.CompletionException;

@Slf4j
@Component
@RequiredArgsConstructor
public class StarRocksStreamLoadClient {

    private final StarRocksProperties properties;
    private final ObjectMapper objectMapper;
    private final BackendSelector backendSelector;
    private final SchemaMapping schemaMapping;
    private final HttpClient httpClient = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(5))
            .followRedirects(HttpClient.Redirect.NEVER).build();

    private HttpRequest buildRequest(URI uri, String csvBody, String label) {
        HttpRequest.Builder builder = HttpRequest.newBuilder().uri(uri).timeout(Duration.ofSeconds(60))
                .header("Authorization", buildAuthorizationHeader())
                .header("format", "CSV")
                .header("column_separator", ",")
                .header("enclose", "\"")
                .header("columns", schemaMapping.starRocksColumns())
                .header("strict_mode", "true")
                .header("max_filter_ratio", "0")
                .header("Content-Type", "text/csv")
                .expectContinue(true);

        if (properties.isMergeCommitEnabled()) {
            builder.header("enable_merge_commit", "true");
            builder.header("merge_commit_async", Boolean.toString(properties.isMergeCommitAsync()));
            builder.header("merge_commit_interval_ms", Long.toString(properties.getMergeCommitIntervalMs()));
            builder.header("merge_commit_parallel", Integer.toString(properties.getMergeCommitParallel()));
        } else {
            builder.header("label", label);
        }
        return builder.PUT(HttpRequest.BodyPublishers.ofString(csvBody, StandardCharsets.UTF_8)).build();
    }

    private String buildStreamLoadUrl() {
        String backend = backendSelector.nextBackend();
        if (backend.endsWith("/")) backend = backend.substring(0, backend.length() - 1);
        return backend + "/api/" + properties.getDatabase() + "/" + properties.getTable() + "/_stream_load";
    }

    public CompletableFuture<StreamLoadResult> loadAsync(List<StarRocksRow> rows, String label) {
        if (rows == null || rows.isEmpty()) {
            return CompletableFuture.failedFuture(new IllegalArgumentException("Stream Load batch cannot be empty"));
        }

        String csvBody = serializeCsv(rows);
        URI uri = URI.create(buildStreamLoadUrl());
        HttpRequest request = buildRequest(uri, csvBody, label);

        return httpClient.sendAsync(request, HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8))
                .thenApply(response -> {
            if (response.statusCode() < 200 || response.statusCode() >= 300) {
                throw new CompletionException(new IOException("StarRocks Stream Load HTTP failure. status="
                        + response.statusCode() + ", body=" + response.body()));
            }
            try {
                return objectMapper.readValue(response.body(), StreamLoadResult.class);
            } catch (IOException e) {
                throw new CompletionException(e);
            }
        });
    }

    private String buildAuthorizationHeader() {
        String credentials = properties.getUsername() + ":" + properties.getPassword();
        return "Basic " + Base64.getEncoder().encodeToString(credentials.getBytes(StandardCharsets.UTF_8));
    }

    private void appendCsvString(StringBuilder csv, String value) {
        csv.append('"');
        for (int i = 0; i < value.length(); i++) {
            char c = value.charAt(i);
            if (c == '"') csv.append("\"\"");
            else csv.append(c);
        }
        csv.append('"');
    }

    private void appendCsvValue(StringBuilder csv, Object value) {
        if (value == null) { csv.append("\\N"); return; }
        if (value instanceof String stringValue) { appendCsvString(csv, stringValue); return; }
        csv.append(value);
    }

    private String serializeCsv(List<StarRocksRow> rows) {
        StringBuilder csv = new StringBuilder(rows.size() * 256);
        for (StarRocksRow row : rows) {
            appendCsvValue(csv, row.kafkaTopic()); csv.append(',');
            appendCsvValue(csv, row.kafkaPartition()); csv.append(',');
            appendCsvValue(csv, row.kafkaOffset()); csv.append(',');
            appendCsvValue(csv, row.kafkaTimestamp());
            for (Object value : row.netFlowValues()) {
                csv.append(',');
                appendCsvValue(csv, value);
            }
            csv.append('\n');
        }
        return csv.toString();
    }
}

package com.nimbo.realtime.consumer;

import com.nimbo.realtime.avro.AvroDeserializer;
import com.nimbo.realtime.client.StarRocksStreamLoadClient;
import com.nimbo.realtime.config.StarRocksProperties;
import com.nimbo.realtime.mapper.StarRocksRecordMapper;
import com.nimbo.realtime.model.StarRocksRow;
import com.nimbo.realtime.model.StreamLoadResult;
import com.nimbo.realtime.validation.NetflowValidationService;
import com.nimbo.realtime.validation.ValidationResult;
import lombok.Getter;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.apache.avro.generic.GenericRecord;
import org.apache.kafka.clients.consumer.ConsumerRecord;
import org.apache.kafka.clients.consumer.ConsumerRecords;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import org.apache.kafka.clients.consumer.OffsetAndMetadata;
import org.apache.kafka.common.TopicPartition;

import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.Deque;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.CompletionException;
import java.util.concurrent.TimeUnit;
import java.util.function.Function;

@Slf4j
@RequiredArgsConstructor
public class StarRocksBatchProcessor {

    private static final String PUBLISH_TIMEOUT = "Publish Timeout";

    private final int workerNumber;
    private final StarRocksProperties starRocksProperties;
    private final AvroDeserializer avroDeserializer;
    private final NetflowValidationService validationService;
    private final StarRocksRecordMapper mapper;
    private final StarRocksStreamLoadClient streamLoadClient;

    private final List<StarRocksRow> rowBatch = new ArrayList<>();
    private final Map<TopicPartition, OffsetAndMetadata> pendingOffsets = new HashMap<>();
    private final Deque<PendingLoad> inFlightLoads = new ArrayDeque<>();
    private long lastFlushTime = System.currentTimeMillis();

    @Getter
    private boolean kafkaPaused;

    private record PendingLoad(
            String label,
            int rowCount,
            Map<TopicPartition, OffsetAndMetadata> offsets,
            CompletableFuture<StreamLoadResult> future
    ) {}

    public void process(ConsumerRecords<byte[], byte[]> records) {
        for (ConsumerRecord<byte[], byte[]> kafkaRecord : records) {
            processRecord(kafkaRecord);
        }
    }

    private void processRecord(ConsumerRecord<byte[], byte[]> kafkaRecord) {
        GenericRecord avroRecord = avroDeserializer.deserialize(kafkaRecord.value());
        ValidationResult validation = validationService.validate(avroRecord);
        markOffsetHandled(kafkaRecord);

        if (!validation.valid()) {
            return;
        }

        StarRocksRow row = mapper.map(
                avroRecord,
                kafkaRecord.topic(),
                kafkaRecord.partition(),
                kafkaRecord.offset(),
                kafkaRecord.timestamp()
        );
        rowBatch.add(row);
    }

    private void markOffsetHandled(ConsumerRecord<byte[], byte[]> kafkaRecord) {
        TopicPartition topicPartition = new TopicPartition(kafkaRecord.topic(), kafkaRecord.partition());
        pendingOffsets.put(topicPartition, new OffsetAndMetadata(kafkaRecord.offset() + 1));
    }

    public void flushIfNeeded() {
        if (!shouldFlush() || rowBatch.isEmpty()) return;

        submitAsync();
        rowBatch.clear();
        pendingOffsets.clear();
        lastFlushTime = System.currentTimeMillis();
    }

    private boolean shouldFlush() {
        return reachedSizeThreshold() || reachedTimeThreshold();
    }

    private boolean reachedSizeThreshold() {
        return rowBatch.size() >= starRocksProperties.getMaxBatchSizeRows();
    }

    private boolean reachedTimeThreshold() {
        if (rowBatch.isEmpty()) return false;
        long elapsed = System.currentTimeMillis() - lastFlushTime;
        return elapsed >= starRocksProperties.getMaxBatchTimeMs();
    }

    private void submitAsync() {
        List<StarRocksRow> rows = List.copyOf(rowBatch);
        Map<TopicPartition, OffsetAndMetadata> offsetsCopy = Map.copyOf(pendingOffsets);
        String label = generateDeterministicLabel(offsetsCopy);
        CompletableFuture<StreamLoadResult> future = loadWithRetryAsync(rows, label);

        inFlightLoads.addLast(new PendingLoad(label, rows.size(), offsetsCopy, future));
        log.debug("Worker {} submitted async Stream Load. label={}, rows={}, inFlight={}",
                workerNumber, label, rows.size(), inFlightLoads.size());
    }

    public void drainCompletedLoads(KafkaConsumer<byte[], byte[]> consumer) {
        while (hasCompletedHead()) {
            PendingLoad pending = inFlightLoads.peekFirst();
            StreamLoadResult result = awaitResult(pending);
            consumer.commitSync(pending.offsets());
            inFlightLoads.removeFirst();
            logCompletedLoad(pending, result);
        }
    }

    private boolean hasCompletedHead() {
        return !inFlightLoads.isEmpty() && inFlightLoads.peekFirst().future().isDone();
    }

    private StreamLoadResult awaitResult(PendingLoad pending) {
        try {
            return pending.future().join();
        } catch (CompletionException exception) {
            log.error("Async StarRocks load failed", exception.getCause());
            throw exception;
        }
    }

    private void logCompletedLoad(PendingLoad pending, StreamLoadResult result) {
        log.info("Worker {} completed Stream Load. label={}, rows={}, status={}, remainingInFlight={}",
                workerNumber, pending.label(), pending.rowCount(), result.status(),     inFlightLoads.size());
    }

    public boolean isBackpressured() {
        return inFlightLoads.size() >= starRocksProperties.getMaxInFlightRequests();
    }

    public void markKafkaPaused() {
        kafkaPaused = true;
    }

    public void markKafkaResumed() {
        kafkaPaused = false;
    }

    public void shutdown(KafkaConsumer<byte[], byte[]> consumer) {
        flushRemainingBatch();
        drainInFlightLoadsOnShutdown(consumer);
    }

    private void flushRemainingBatch() {
        if (rowBatch.isEmpty()) return;
        submitAsync();
        rowBatch.clear();
        pendingOffsets.clear();
    }

    private void drainInFlightLoadsOnShutdown(KafkaConsumer<byte[], byte[]> consumer) {
        while (!inFlightLoads.isEmpty()) {
            PendingLoad pending = inFlightLoads.peekFirst();
            try {
                pending.future().join();
                consumer.commitSync(pending.offsets());
                inFlightLoads.removeFirst();
            } catch (CompletionException exception) {
                log.error("Failed to complete Stream Load during shutdown", exception.getCause());
                return;
            }
        }
    }

    private CompletableFuture<StreamLoadResult> loadWithRetryAsync(List<StarRocksRow> rows, String label) {
        return streamLoadClient.loadAsync(rows, label)
                .handle((result, error) -> handleLoadResult(rows, label, result, error))
                .thenCompose(Function.identity());
    }

    private CompletableFuture<StreamLoadResult> handleLoadResult(List<StarRocksRow> rows
            , String label, StreamLoadResult result, Throwable error) {
        if (error == null && isSuccessfulLoad(result)) return CompletableFuture.completedFuture(result);
        logLoadFailure(result, error, label);
        return retryLater(rows, label);
    }

    private boolean isSuccessfulLoad(StreamLoadResult result) {
        return result.isSuccessful() || PUBLISH_TIMEOUT.equalsIgnoreCase(result.status());
    }

    private void logLoadFailure(StreamLoadResult result, Throwable error, String label) {
        if (error != null) {
            log.error("Worker {} Stream Load HTTP failure. label={}. Retrying...", workerNumber, label, error);
            return;
        }
        log.error("Worker {} Stream Load failed. status={}, message={}, label={}. Retrying...",
                workerNumber, result.status(), result.message(), label);
    }

    private CompletableFuture<StreamLoadResult> retryLater(List<StarRocksRow> rows, String label) {
        return delay(starRocksProperties.getRetryBackoffMs()).thenCompose(ignored -> loadWithRetryAsync(rows, label));
    }

    private CompletableFuture<Void> delay(long milliseconds) {
        return CompletableFuture.runAsync(() -> {}, CompletableFuture.delayedExecutor(milliseconds, TimeUnit.MILLISECONDS));
    }

    private String generateDeterministicLabel(Map<TopicPartition, OffsetAndMetadata> offsets) {
        StringBuilder builder = new StringBuilder("nimbo-w").append(workerNumber);
        offsets.entrySet().stream()
                .sorted(Comparator.comparingInt(entry -> entry.getKey().partition()))
                .forEach(entry -> appendOffsetToLabel(builder, entry));
        return builder.toString();
    }

    private void appendOffsetToLabel(StringBuilder builder, Map.Entry<TopicPartition, OffsetAndMetadata> entry) {
        builder.append("-p").append(entry.getKey().partition()).append("-o").append(entry.getValue().offset());
    }
}

package com.nimbo.realtime.consumer;

import com.nimbo.realtime.avro.AvroDeserializer;
import com.nimbo.realtime.client.StarRocksStreamLoadClient;
import com.nimbo.realtime.config.KafkaProperties;
import com.nimbo.realtime.config.StarRocksProperties;
import com.nimbo.realtime.mapper.StarRocksRecordMapper;
import com.nimbo.realtime.validation.NetflowValidationService;
import jakarta.annotation.PostConstruct;
import jakarta.annotation.PreDestroy;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import org.springframework.scheduling.concurrent.CustomizableThreadFactory;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

@Slf4j
@Component
@RequiredArgsConstructor
public class StarRocksConsumerManager {

    private final KafkaProperties kafkaProperties;
    private final StarRocksProperties starRocksProperties;
    private final KafkaConsumerFactory consumerFactory;
    private final AvroDeserializer avroDeserializer;
    private final NetflowValidationService validationService;
    private final StarRocksRecordMapper mapper;
    private final StarRocksStreamLoadClient streamLoadClient;
    private final List<StarRocksConsumerWorker> workers = new ArrayList<>();
    private ExecutorService executor;

    @PostConstruct
    public void start() {
        int consumerCount = kafkaProperties.getConsumerCount();
        CustomizableThreadFactory threadFactory = new CustomizableThreadFactory("starrocks-worker-");
        executor = Executors.newFixedThreadPool(consumerCount, threadFactory);

        for (int workerNumber = 1; workerNumber <= consumerCount; workerNumber++) {
            startWorker(workerNumber);
        }
        log.info("Started {} StarRocks Kafka consumer workers", consumerCount);
    }

    private void startWorker(int workerNumber) {
        KafkaConsumer<byte[], byte[]> consumer = consumerFactory.create(workerNumber, kafkaProperties);
        StarRocksBatchProcessor batchProcessor = new StarRocksBatchProcessor(workerNumber, starRocksProperties,
                avroDeserializer, validationService, mapper, streamLoadClient);
        StarRocksConsumerWorker worker = new StarRocksConsumerWorker(workerNumber, consumer, kafkaProperties, batchProcessor);

        workers.add(worker);
        executor.submit(worker);
    }

    @PreDestroy
    public void stop() {
        log.info("Stopping {} StarRocks Kafka consumer workers", workers.size());
        workers.forEach(StarRocksConsumerWorker::stop);
        shutdownExecutor();
    }

    private void shutdownExecutor() {
        if (executor == null) return;
        executor.shutdown();
        try {
            if (!executor.awaitTermination(10, TimeUnit.SECONDS)) {
                log.warn("Workers did not shut down gracefully in time, forcing shutdown...");
                executor.shutdownNow();
            }
        } catch (InterruptedException exception) {
            log.error("Interrupted during worker shutdown, forcing termination", exception);
            executor.shutdownNow();
            Thread.currentThread().interrupt();
        }
    }
}

package com.nimbo.realtime.consumer;

import com.nimbo.realtime.config.KafkaProperties;
import com.nimbo.realtime.config.StarRocksProperties;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.apache.kafka.clients.consumer.ConsumerRecords;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import org.apache.kafka.common.errors.WakeupException;

import java.time.Duration;
import java.util.List;

@Slf4j
@RequiredArgsConstructor
public class StarRocksConsumerWorker implements Runnable {

    private final int workerNumber;
    private final KafkaConsumer<byte[], byte[]> consumer;
    private final KafkaProperties kafkaProperties;
    private final StarRocksBatchProcessor batchProcessor;
    private volatile boolean running = true;

    @Override
    public void run() {
        try {
            consumer.subscribe(List.of(kafkaProperties.getTopic()));
            log.info("StarRocks consumer worker {} started", workerNumber);

            while (running) {
                processIteration();
            }
        } catch (WakeupException exception) {
            if (running) {
                log.error("Worker {} received unexpected wakeup", workerNumber, exception);
            }
        } catch (Exception exception) {
            log.error("Worker {} stopped due to unexpected error", workerNumber, exception);
        } finally {
            closeConsumer();
        }
    }

    private void processIteration() {
        batchProcessor.drainCompletedLoads(consumer);

        if (batchProcessor.isBackpressured()) {
            pauseKafka();
            consumer.poll(Duration.ofMillis(100));
            return;
        }

        resumeKafka();
        ConsumerRecords<byte[], byte[]> records = consumer.poll(Duration.ofMillis(kafkaProperties.getPollTimeoutMs()));
        batchProcessor.process(records);
        batchProcessor.flushIfNeeded();
    }

    private void pauseKafka() {
        if (consumer.assignment().isEmpty() || batchProcessor.isKafkaPaused()) return;
        consumer.pause(consumer.assignment());
        batchProcessor.markKafkaPaused();
        log.debug("Worker {} paused Kafka", workerNumber);
    }

    private void resumeKafka() {
        if (!batchProcessor.isKafkaPaused()) return;
        consumer.resume(consumer.assignment());
        batchProcessor.markKafkaResumed();
        log.debug("Worker {} resumed Kafka", workerNumber);
    }

    private void closeConsumer() {
        try {
            batchProcessor.shutdown(consumer);
        } finally {
            consumer.close();
            log.info("StarRocks consumer worker {} stopped", workerNumber);
        }
    }

    public void stop() {
        running = false;
        consumer.wakeup();
    }
}

package com.nimbo.realtime.mapper;

import org.apache.avro.Schema;
import org.apache.avro.generic.GenericRecord;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.Locale;

@Component
public class SchemaMapping {

    private static final List<String> KAFKA_COLUMNS = List.of(
            "kafka_topic",
            "kafka_partition",
            "kafka_offset",
            "kafka_timestamp"
    );

    private final List<Schema.Field> fields;
    private final String starRocksColumns;

    public SchemaMapping(Schema schema) {
        this.fields = List.copyOf(schema.getFields());

        List<String> columns = new ArrayList<>(KAFKA_COLUMNS.size() + fields.size());
        columns.addAll(KAFKA_COLUMNS);

        fields.stream()
                .map(Schema.Field::name)
                .map(name -> name.toLowerCase(Locale.ROOT))
                .forEach(columns::add);

        this.starRocksColumns = String.join(",", columns);
    }

    public String starRocksColumns() {
        return starRocksColumns;
    }

    public List<Object> extractValues(GenericRecord genericRecord) {
        List<Object> values = new ArrayList<>(fields.size());
        for (Schema.Field field : fields) {
            Object value = genericRecord.get(field.name());
            values.add(normalizeValue(value, field.schema()));
        }
        return values;
    }

    private Object normalizeValue(Object value, Schema fieldSchema) {
        if (value == null) return null;

        return switch (fieldSchema.getType()) {
            case STRING -> value.toString();
            case INT -> ((Number) value).intValue();
            case LONG -> ((Number) value).longValue();
            case DOUBLE -> ((Number) value).doubleValue();
            case FLOAT -> ((Number) value).floatValue();
            default -> value;
        };
    }
}

package com.nimbo.realtime.validation;

import org.apache.avro.generic.GenericRecord;
import org.springframework.stereotype.Service;

@Service
public class NetflowValidationService {

    private static final String FLOW_DURATION_MILLISECONDS = "FLOW_DURATION_MILLISECONDS";

    public ValidationResult validate(GenericRecord genericRecord) {
        if (genericRecord == null) {
            return ValidationResult.failure("Record is null");
        }

        StringBuilder errors = new StringBuilder();

        checkIpv4(genericRecord, "IPV4_SRC_ADDR", errors);
        checkIpv4(genericRecord, "IPV4_DST_ADDR", errors);

        checkRange(genericRecord, "L4_SRC_PORT", 0, 65_535, errors);
        checkRange(genericRecord, "L4_DST_PORT", 0, 65_535, errors);
        checkRange(genericRecord, "PROTOCOL", 0, 255, errors);

        checkRange(genericRecord, "MIN_TTL", 0, 255, errors);
        checkRange(genericRecord, "MAX_TTL", 0, 255, errors);

        checkRange(genericRecord, "MIN_IP_PKT_LEN", 0, 65_535, errors);
        checkRange(genericRecord, "MAX_IP_PKT_LEN", 0, 65_535, errors);

        checkRange(genericRecord, "ICMP_TYPE", 0, 65_535, errors);
        checkRange(genericRecord, "ICMP_IPV4_TYPE", 0, 255, errors);

        checkRange(genericRecord, "DNS_QUERY_ID", 0, 65_535, errors);
        checkRange(genericRecord, "DNS_QUERY_TYPE", 0, 65_535, errors);

        checkNonNegative(genericRecord, "IN_BYTES", errors);
        checkNonNegative(genericRecord, "OUT_BYTES", errors);
        checkNonNegative(genericRecord, "IN_PKTS", errors);
        checkNonNegative(genericRecord, "OUT_PKTS", errors);

        checkNonNegative(genericRecord, FLOW_DURATION_MILLISECONDS, errors);

        checkNonNegative(genericRecord, "DURATION_IN", errors);
        checkNonNegative(genericRecord, "DURATION_OUT", errors);

        checkNonNegative(genericRecord, "LONGEST_FLOW_PKT", errors);
        checkNonNegative(genericRecord, "SHORTEST_FLOW_PKT", errors);

        checkMinMax(genericRecord, "MIN_TTL", "MAX_TTL", errors);
        checkMinMax(genericRecord, "SHORTEST_FLOW_PKT", "LONGEST_FLOW_PKT", errors);
        checkMinMax(genericRecord, "MIN_IP_PKT_LEN", "MAX_IP_PKT_LEN", errors);
        checkMinMax(genericRecord, "DURATION_OUT", FLOW_DURATION_MILLISECONDS, errors);
        checkMinMax(genericRecord, "DURATION_IN", FLOW_DURATION_MILLISECONDS, errors);

        checkFiniteNonNegativeDouble(genericRecord, "L7_PROTO", errors);

        return errors.isEmpty() ? ValidationResult.success() : ValidationResult.failure(errors.toString());
    }

    private void checkNonNegative(GenericRecord genericRecord, String field, StringBuilder errors) {
        Number value = number(genericRecord, field, errors);
        if (value != null && value.longValue() < 0) {
            addError(errors, field + " is negative: " + value);
        }
    }

    private void checkRange(GenericRecord genericRecord, String field, long min, long max, StringBuilder errors) {
        Number value = number(genericRecord, field, errors);
        if (value == null) return;

        long number = value.longValue();
        if (number < min || number > max) {
            addError(errors, field + " out of range [" + min + ", " + max + "]: " + number);
        }
    }

    private void checkMinMax(GenericRecord genericRecord, String minField, String maxField, StringBuilder errors) {
        Object minValue = genericRecord.get(minField);
        Object maxValue = genericRecord.get(maxField);

        if (!(minValue instanceof Number min) || !(maxValue instanceof Number max)) return;

        if (min.longValue() > max.longValue()) {
            addError(errors, minField + " > " + maxField + " (" + min + " > " + max + ")");
        }
    }

    private void checkFiniteNonNegativeDouble(GenericRecord genericRecord, String field, StringBuilder errors) {
        Number value = number(genericRecord, field, errors);
        if (value == null) return;

        double number = value.doubleValue();
        if (!Double.isFinite(number)) {
            addError(errors, field + " is not finite: " + number);
        } else if (number < 0) {
            addError(errors, field + " is negative: " + number);
        }
    }

    private Number number(GenericRecord genericRecord, String field, StringBuilder errors) {
        Object value = genericRecord.get(field);
        if (value == null) {
            addError(errors, field + " is null");
            return null;
        }

        if (!(value instanceof Number number)) {
            addError(errors, field + " is not numeric");
            return null;
        }

        return number;
    }

    private void checkIpv4(GenericRecord genericRecord, String field, StringBuilder errors) {
        Object value = genericRecord.get(field);
        if (value == null) {
            addError(errors, field + " is null");
            return;
        }

        String ip = value.toString();
        String[] parts = ip.split("\\.", -1);
        if (parts.length != 4) {
            addInvalidIpv4(errors, field, ip);
            return;
        }

        for (String part : parts) {
            if (part.isEmpty() || part.length() > 3 || !part.chars().allMatch(Character::isDigit)) {
                addInvalidIpv4(errors, field, ip);
                return;
            }
            if (Integer.parseInt(part) > 255) {
                addInvalidIpv4(errors, field, ip);
                return;
            }
        }
    }

    private void addInvalidIpv4(StringBuilder errors, String field, String ip) {
        addError(errors, field + " invalid IPv4: " + ip);
    }

    private void addError(StringBuilder errors, String error) {
        if (!errors.isEmpty()) errors.append(" | ");
        errors.append(error);
    }
}
